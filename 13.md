      "Download Facebook videos",

      "MP4 with original audio",
      "No watermark added",
      "No app or extension",
    ],
  };

  const faq = {
    "@context": "https://schema.org",
    "@type": "FAQPage",
    mainEntity: FAQ.map((f) => ({
      "@type": "Question",
      name: f.q,
      acceptedAnswer: { "@type": "Answer", text: f.a },
    })),
  };

  return (
    <>
      <script type="application/ld+json" dangerouslySetInnerHTML={{ __html: JSON.stringify(webApp) }} />

四、这页刻意没做的四件事（都记进档案）
不为 without watermark 开第二页：实测第 8 名 fbdown.blog 已经是标题精确匹配的专门页（.blog 域名），而它的量来自「需求方式变了」而不是「新需求出现」——我们用一个 H2 + 一条 FAQ + 一条徽章承接，位置够了，第二页只会和主页自竞争。
不写 facebook downloader(9,140,000)、不为 facebook video download(20,400,000) 改 Title：前者出价 $0.01~$0.07 是导航意图，后者与主词同价同形是合并组的整组数。
不标 1080p / 2K / 4K：第 9 名 snapsave.app 在标，那是虚标路线（720p 上传不可能变 4K），我们写「跟随原上传、不做放大」，并专门有一条 FAQ 解释「为什么下载看着比屏幕上软」——这条真实困惑是对手普遍没答的。
不做 /facebook-watch-video-downloader/：140/月，无量，只在 TYPES 里带一句 Watch 长视频。
