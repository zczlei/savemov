"use client";

import { useEffect, useRef, useState } from "react";

type Option = { label: string; format: "MP4" | "MP3"; url: string; sizeMb?: number };
type Job = {
  status: "queued" | "fetching" | "preparing" | "ready" | "error";
  message?: string;
  title?: string;
  options?: Option[];
};

const STEPS = ["URL", "Fetch", "Prepare", "Ready"] as const;
const IDX = { queued: 0, fetching: 1, preparing: 2, ready: 3, error: 0 } as const;

const MSG = {
  badUrl: "That doesn't look like a link. Paste the full URL of a public video page (it must start with https://).",
  network: "Couldn't reach Savemov. Check your connection and try again.",
  unsupported:
    "We couldn't find a downloadable video on that page. Private posts, login-only pages and DRM streams can't be saved.",
};

function isHttpUrl(v: string) {
  if (!v || v.length > 2048) return false;
  try {
    const u = new URL(v);
    return (u.protocol === "http:" || u.protocol === "https:") && u.hostname.includes(".");
  } catch {
    return false;
  }
}

export default function DownloaderForm() {
  const [url, setUrl] = useState("");
  const [job, setJob] = useState<Job | null>(null);
  const [busy, setBusy] = useState(false);
  const timer = useRef<number | null>(null);

  const stop = () => { if (timer.current) { clearTimeout(timer.current); timer.current = null; } };
  useEffect(() => () => stop(), []);

  async function poll(id: string) {
    try {
      const res = await fetch(`/api/download?job=${encodeURIComponent(id)}`);
      const data: Job = await res.json();
      setJob(data);
      if (data.status === "ready" || data.status === "error") { stop(); setBusy(false); return; }
      timer.current = window.setTimeout(() => poll(id), 1200);
    } catch {
      stop(); setBusy(false); setJob({ status: "error", message: MSG.network });
    }
  }

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    const v = url.trim();
    if (!isHttpUrl(v)) { setJob({ status: "error", message: MSG.badUrl }); return; }
    stop();
    setBusy(true);
    setJob({ status: "queued" });
    try {
      const res = await fetch("/api/download", {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ url: v }),
      });
      const data = (await res.json()) as { job?: string; message?: string };
      if (!res.ok || !data.job) {
        setBusy(false);
        setJob({ status: "error", message: data.message ?? MSG.unsupported });
        return;
      }
      poll(data.job);
    } catch {
      setBusy(false);
      setJob({ status: "error", message: MSG.network });
    }
  }

  async function paste() {
    try {
      const t = await navigator.clipboard.readText();
      if (t) setUrl(t.trim());
    } catch { /* 用户拒绝剪贴板权限：不报错，让他手动粘贴 */ }
  }

  const step = job ? IDX[job.status] : 0;
  const ready = job?.status === "ready";

  return (
    <div className="rounded-xl border border-slate-200 bg-white p-4 shadow-sm">
      <form onSubmit={submit} noValidate>
        <label htmlFor="video-url" className="block text-sm font-medium text-slate-700">
          Video URL or link
        </label>
        <div className="mt-2 flex flex-col gap-2 sm:flex-row">
          <input
            id="video-url"
            ref={undefined}
            type="url"
            inputMode="url"
            name="url"
            autoComplete="off"
            spellCheck={false}
            value={url}
            onChange={(e) => setUrl(e.target.value)}
            placeholder="Paste a TikTok, Instagram, Facebook, Telegram or Pinterest link"
            aria-describedby="url-status"
            className="min-w-0 flex-1 rounded-lg border border-slate-300 px-4 py-3 text-base outline-none focus:border-slate-900 focus:ring-2 focus:ring-slate-900/10"
          />
          <div className="flex gap-2">
            <button type="button" onClick={paste}
              className="rounded-lg border border-slate-300 px-4 py-3 text-sm font-medium hover:bg-slate-50">
              Paste
            </button>
            <button type="submit" disabled={busy}
              className="flex-1 rounded-lg bg-slate-900 px-6 py-3 text-base font-semibold text-white disabled:opacity-60 sm:w-auto">
              {busy ? "Working…" : "Download MP4"}
            </button>
          </div>
        </div>
        {url && !busy && (
          <button type="button" onClick={() => { setUrl(""); setJob(null); stop(); }}
            className="mt-2 text-sm text-slate-500 underline">Clear</button>
        )}
      </form>

      {/* 进度条：常驻，避免状态出现时页面跳动 */}
      <ol className="mt-4 flex items-center gap-2 text-xs font-medium" aria-label="Download progress">
        {STEPS.map((s, i) => (
          <li key={s} className="flex items-center gap-2">
            <span
              aria-current={i === step ? "step" : undefined}
              className={i === step && !ready ? "rounded-full bg-slate-900 px-2.5 py-1 text-white"
                : i < step || ready ? "rounded-full bg-slate-200 px-2.5 py-1 text-slate-700"
                : "rounded-full px-2.5 py-1 text-slate-400"}
            >
              {i + 1} {s}
            </span>
            {i < STEPS.length - 1 && <span className="text-slate-300">›</span>}
          </li>
        ))}
      </ol>

      <p id="url-status" role="status" aria-live="polite" className="mt-3 min-h-[1.5rem] text-sm text-slate-600">
        {job?.message ?? (job && !ready ? "Fetching the video…" : null)}

          {job.title && <p className="truncate text-sm font-semibold">{job.title}</p>}
          <ul className="mt-2 space-y-2">
            {job.options.map((o) => (
              <li key={o.url} className="flex items-center justify-between gap-3 text-sm">
                <span className="font-medium">{o.label}{o.sizeMb ? ` · ${o.sizeMb} MB` : ""}</span>
                <a href={o.url} download rel="noopener nofollow"
                   className="rounded-md bg-slate-900 px-4 py-2 font-semibold text-white">
                  Save {o.format}
                </a>
              </li>
            ))}
          </ul>
        </div>
      ) : null}
    </div>
  );
}

import { NextResponse } from "next/server";

export const runtime = "nodejs"; // yt-dlp / fluent-ffmpeg 只能跑在 Node 运行时，别用 edge

type JobState = {
  id: string; url: string; status: "queued" | "fetching" | "preparing" | "ready" | "error";
  title?: string; options?: { label: string; format: "MP4" | "MP3"; url: string; sizeMb?: number }[];
  message?: string; createdAt: number;
};

/* ⚠️ 必改：内存 Map 只在单实例有效。Vercel/多实例部署必须换 Redis 或 DB，
   否则轮询打到另一个实例，任务永远停在 Prepare。 */
const jobs = new Map<string, JobState>();
const TTL_MS = 15 * 60_000;

const RATE = new Map<string, { n: number; t: number }>();
function tooMany(ip: string) {
  const now = Date.now();
  const hit = RATE.get(ip);
  if (!hit || now - hit.t > 60_000) { RATE.set(ip, { n: 1, t: now }); return false; }
  hit.n += 1;
  return hit.n > 6; // ⚠️ 必改：按你的成本定，6 次/分钟
}

function sanitize(raw: unknown): string | null {
  if (typeof raw !== "string" || raw.length > 2048) return null;
  try {
    const u = new URL(raw.trim());
    if (u.protocol !== "https:" && u.protocol !== "http:") return null;
    if (!u.hostname.includes(".")) return null;
    return u.toString();
  } catch { return null; }
}

export async function POST(req: Request) {
  const ip = req.headers.get("x-forwarded-for")?.split(",")[0]?.trim() ?? "unknown";
  if (tooMany(ip)) {
    return NextResponse.json({ message: "Too many requests — wait a minute and try again." }, { status: 429 });
  }
  const body = (await req.json().catch(() => ({}))) as { url?: unknown };
  const url = sanitize(body.url);
  if (!url) return NextResponse.json({ message: "Paste a full public video URL." }, { status: 400 });

  const id = crypto.randomUUID();
  const job: JobState = { id, url, status: "queued", createdAt: Date.now() };
  jobs.set(id, job);

  resolve(job).catch(() => {}); // 不 await：前端轮询状态，避免长请求被网关切断
  return NextResponse.json({ job: id, status: "queued" });
}

export async function GET(req: Request) {
  const id = new URL(req.url).searchParams.get("job");
  const job = id ? jobs.get(id) : undefined;
  if (!job) return NextResponse.json({ status: "error", message: "That job is gone. Paste the link again." }, { status: 404 });
  const { url, ...safe } = job; // 不把用户粘贴的原始链接回传前端
  void url;
  return NextResponse.json(safe);
}

/**
 * ⚠️ 必改：这里换成真实实现。参考做法（Node 侧调 yt-dlp）：
 *   yt-dlp -J --no-playlist <url>        → 拿 title / formats / thumbnail
 *   取最佳 mp4 与最佳音频，音频再 ffmpeg 转 mp3
 *   自己代理下载或推到你自己的对象存储，再把 URL 交给前端
 * 关键坑：平台给的 signed CDN URL 通常几十分钟就失效，别把 yt-dlp 返回的
 * 原始 url 直接塞给前端——先落到自己的存储/代理，否则用户点了报 403。
 */
async function resolve(job: JobState) {
  job.status = "fetching";
  // await probePlatformUrl(job.url)
  job.status = "preparing";
  // await transcodeAndStore(...)
  job.title = "Video";
  job.options = [
    { label: "MP4 · 720p", format: "MP4", url: "/demo.mp4", sizeMb: 12 },
    { label: "MP4 · 480p", format: "MP4", url: "/demo.mp4", sizeMb: 6 },
    { label: "MP3 · audio", format: "MP3", url: "/demo.mp3", sizeMb: 2 },
  ];
  job.status = "ready";
}

// 回收过期任务（生产环境交给 Redis TTL 或定时清理，别用这个 setInterval）
setInterval(() => {
  const now = Date.now();
  for (const [id, j] of jobs) if (now - j.createdAt > TTL_MS) jobs.delete(id);
}, 60_000).unref?.();
