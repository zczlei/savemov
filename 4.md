export function buildMetadata(c: Copy): Metadata {
  return {
    title: { absolute: c.title },
    description: c.desc,
    alternates: {
      canonical: c.path,
      languages: { en: SITE, pt: `${SITE}/pt/`, es: `${SITE}/es/`, "x-default": SITE },
    },
    openGraph: {
      type: "website", url: SITE + c.path, siteName: "Savemov", locale: c.ogLocale,
      alternateLocale: ["en_US", "pt_BR", "es_ES"].filter((l) => l !== c.ogLocale),
      title: c.title, description: c.desc,
      images: [{ url: "/og.png", width: 1200, height: 630, alt: c.h1 }], // ⚠️ 必改：放一张真图
    },
    twitter: { card: "summary_large_image", title: c.title, description: c.desc, images: ["/og.png"] },
    robots: { index: true, follow: true, maxSnippet: -1, maxImagePreview: "large" },
  };
}

  return [
    {
      "@context": "https://schema.org", "@type": "WebApplication",
      name: "Savemov – " + c.h1, url: SITE + c.path, description: c.desc,
      applicationCategory: "UtilityApplication", operatingSystem: "Any (web browser)", inLanguage: c.lang,
      isAccessibleForFree: true,
      hasPart: { "@type": "Offer", price: "0", priceCurrency: "USD", description: c.quota },
      featureList: [c.h1, "MP4 y MP3", c.quality, c.trust],
    },
    {
      "@context": "https://schema.org", "@type": "FAQPage",
      mainEntity: c.faqs.map((f) => ({
        "@type": "Question", name: f.q, acceptedAnswer: { "@type": "Answer", text: f.a },
      })),
    },
  ];
}

拷贝后必须处理的 4 件事
DownloaderForm 我还没给实现（上一轮只写了引用）。现在它需要的 props 是 copy（= c.form：label / placeholder / button / paste / clear / steps / security / busy / error / formats）加 quota、quality。要我下一步把这个客户端组件补全（粘贴即取、拖拽、结果卡、清晰度多选、失败态），说一声就行。
内链 404 风险：PP = "" 让两版首页的平台链接都指向现存的英文平台页（不会 404）。你做好 /pt/tiktok-downloader/ 那批之后，把 PT 的 PP 改成 "/pt"、ES 改成 "/es"——跨语言内链是语言版最常见的失误，改了以后页脚那 11 个链接会一起切过去，所以先确认目标页存在。
<html lang> 的限制：App Router 里只有根 layout 能渲染 <html>，所以根 layout 只能是 lang="en"。我用 <main lang="pt"> / lang="es" 补页面级语言，加上 hreflang、og:locale、alternateLocale，这三层足够让谷歌判定语言版（盘面上 snapwc 的 /pt 也不是靠独立 html 标签）。如果以后语言做到 5 种以上、要 100% 干净，就得把根 layout 改成 passthrough、用 app/[lang]/layout.tsx 渲染 <html>，那要动你现在的英文版结构，等你开口再改。
[N1]/[N2]/[N3]：组件里有一行开发期警告，占位符没替换会在终端打日志、拦住你误发。另外 /terms /copyright /privacy /pricing 这四个路由得存在（首页有链接，断链比没链接差）。
📌 sitemap.xml 记得加上这两条，robots.txt 别屏蔽 /pt /es——新语言版最容易卡在「有 hreflang 但 sitemap 里没有，谷歌不知道去抓」。
