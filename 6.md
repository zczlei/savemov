"use client";

import { useRef, useState } from "react";

type Props = {
  label?: string; placeholder?: string; button?: string; paste?: string;
  clear?: string; busy?: string; done?: string; error?: string; formats?: string;
  api?: string;
};

type Row = { quality?: string; format: string; size?: string; link: string };

export default function DownloaderForm({
  label = "URL or link to the video",
  placeholder = "Paste a TikTok, Instagram, Facebook, Pinterest or Telegram link",
  button = "Download MP4",
  paste = "Paste",
  clear = "Clear",
  busy = "Finding the video…",
  done = "Ready – choose a file",
  error = "No video found at that link. Make sure the page is public and try again.",
  formats = "MP4 or MP3",
  api = "/api/download",
}: Props) {
  const [url, setUrl] = useState("");
  const [status, setStatus] = useState<"idle" | "work" | "ok" | "err">("idle");
  const [rows, setRows] = useState<Row[]>([]);
  const [msg, setMsg] = useState("");
  const input = useRef<HTMLInputElement>(null);

  const STEP_LABELS = ["Link", "Search", "Prepare", "Ready"];
  const activeStep = status === "idle" ? 0 : status === "work" ? 2 : status === "ok" ? 3 : 0;

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    const v = url.trim();
    if (!/^https?:\/\/.+\..+/i.test(v)) { setMsg("Paste a full public link (starting with http:// or https://)."); setStatus("err"); return; }
    setStatus("work"); setRows([]); setMsg("");
    try {
      const ctl = new AbortController();
      const timer = setTimeout(() => ctl.abort(), 20000);
      const r = await fetch(api, {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ url: v }), signal: ctl.signal,
      });
      clearTimeout(timer);
      const j = await r.json();
      if (!r.ok || !Array.isArray(j.formats) || j.formats.length === 0) {
        setMsg(typeof j?.error === "string" ? j.error : error); setStatus("err"); return;
      }
      setRows(j.formats as Row[]); setStatus("ok");
    } catch {
      setMsg(error); setStatus("err");
    }
  }

  async function doPaste() {
    try { const t = await navigator.clipboard.readText(); if (t) { setUrl(t); input.current?.focus(); } }
    catch { input.current?.focus(); } // iOS/无权限：退回让用户手动粘贴
  }

  return (
    <div className="rounded-xl border border-slate-200 bg-white p-4 shadow-sm">
      {/* 进度条：工具站的"感知完成度"，直接影响停留与复访 */}
      <ol className="mb-3 flex items-center gap-1 text-[11px] font-medium text-slate-400">
        {STEP_LABELS.map((s, i) => (
          <li key={s} className="flex items-center gap-1">
            <span className={i === activeStep ? "rounded bg-slate-900 px-2 py-0.5 text-white" : "px-2 py-0.5"}>
              {i + 1} {s}
            </span>
            {i < 3 && <span aria-hidden>›</span>}
          </li>
        ))}
      </ol>

      <form onSubmit={submit} noValidate>
        <label htmlFor="vid" className="block text-sm font-medium text-slate-700">{label}</label>
        <div className="mt-2 flex flex-col gap-2 sm:flex-row">
          <input
            id="vid" ref={input} value={url}
            onChange={(e) => { setUrl(e.target.value); if (status === "err") setStatus("idle"); }}
            placeholder={placeholder} type="url" inputMode="url" name="url"
            autoComplete="off" spellCheck={false}
            className="min-w-0 flex-1 rounded-md border border-slate-300 px-3 py-2.5 text-sm outline-none focus:border-slate-900"
          />
          <button type="submit" disabled={status === "work"}
            className="rounded-md bg-slate-900 px-5 py-2.5 text-sm font-semibold text-white disabled:opacity-60">
            {status === "work" ? busy : button}
          </button>
        </div>
        <div className="mt-2 flex items-center gap-3 text-xs text-slate-500">
          <button type="button" onClick={doPaste} className="underline">{paste}</button>
          <button type="button" onClick={() => { setUrl(""); setRows([]); setStatus("idle"); }} className="underline">{clear}</button>
          <span>{formats}</span>
        </div>
      </form>

      <div aria-live="polite">
        {status === "err" && <p className="mt-3 rounded-md bg-red-50 px-3 py-2 text-sm text-red-700">{msg}</p>}
        {status === "ok" && (
          <div className="mt-4">
            <p className="text-sm font-semibold text-slate-700">{done}</p>
            <ul className="mt-2 divide-y divide-slate-100">
              {rows.map((r, i) => (
                <li key={i} className="flex items-center justify-between gap-3 py-2 text-sm">
                  <span className="text-slate-700">
                    {r.format.toUpperCase()}{r.quality ? ` · ${r.quality}` : ""}{r.size ? ` · ${r.size}` : ""}
                  </span>
                  <a href={r.link} download rel="noopener noreferrer nofollow"
                     className="shrink-0 rounded border border-slate-300 px-3 py-1 font-medium hover:bg-slate-50">Save</a>
                </li>
              ))}
            </ul>
          </div>
        )}
      </div>
    </div>
  );
}
拷完之后必做的 4 件事
/api/download 契约：POST {url} → {formats:[{format:"mp4",quality:"1080p",size:"12.4 MB",link:"https://..."}]}；失败返回非 2xx + {error:"…中文/英文照实写"}。没做好这个接口之前别放出来——表单一直报错的页面，排名只会掉得更快。
og.png 要重新出：上一版图上的大字写着 "Download Video from URL"，那是图片，改代码改不到它，按新 Title 改成 Download Video from Link or URL。
tsconfig.json 要有 "paths": { "@/*": ["./*"] }；/pro /terms /privacy /copyright 四个链接要么建页要么删。
部署后验一遍是不是真静态：curl -s https://savemov.com/ | grep -o '<h1[^>]*>[^<]*' 要能打出 Download Videos from Any Link or URL——打不出来说明页面被客户端渲染了，谷歌首轮抓不到正文，前面所有文案工作都白费。
