/* hreflang 只发这个平台“真的建了”的语言版——小平台 locales:["en"] 就只输出自己，
   绝不会给不存在的 /pt/ 页发声明（那是上一轮 Twitter 那类 bug 的源头）。 */
export function buildMetadata(copy: Copy, group: Locale[]): Metadata {
  const url = pageUrl(copy.path, copy.locale);
  const langs: Record<string, string> = {};
  for (const l of group) langs[l === "en" ? "x-default" : l] = pageUrl(copy.path, l);
  return {
    title: copy.title,
    description: copy.description,
    alternates: { canonical: url, languages: langs },
    openGraph: {
      url, type: "website", locale: copy.ogLocale,
      siteName: "Savemov", title: copy.title, description: copy.description,
    },
    twitter: { card: "summary_large_image", title: copy.title, description: copy.description },
  };
}

function jsonLd(copy: Copy, group: Locale[]) {
  const url = pageUrl(copy.path, copy.locale);
  return [
    {
      "@context": "https://schema.org", "@type": "WebApplication",
      name: copy.appName, url,
      applicationCategory: "MultimediaApplication", operatingSystem: "Any",
      description: copy.description, featureList: copy.featureList,
      offers: { "@type": "Offer", price: "0", priceCurrency: "USD" },
    },
    {
      "@context": "https://schema.org", "@type": "FAQPage",
      mainEntity: copy.faq.items.map((i) => ({
        "@type": "Question", name: i.q,
        acceptedAnswer: { "@type": "Answer", text: i.a },
      })),
    },
    {
      "@context": "https://schema.org", "@type": "BreadcrumbList",
      itemListElement: [
        { "@type": "ListItem", position: 1, name: copy.home, item: SITE + (copy.locale === "en" ? "/" : `/${copy.locale}/`) },
        { "@type": "ListItem", position: 2, name: copy.crumb, item: url },
      ],
    },
  ].map((o) => (
    <script key={o["@type"]} type="application/ld+json"
      dangerouslySetInnerHTML={{ __html: JSON.stringify(o) }} />
  ));
}

import { Copy, pageUrl, Locale } from "@/lib/copyTypes";

/* 档案定的口径：平台页底部 4 条同区同语言互链，不做全站互链。

export default function RelatedLinks({
  copy, title, items,

  copy: Copy;

        <ul className="mt-4 flex flex-wrap gap-3">
          {items.map((i) => (
            <li key={i.slug}>
              <a
                href={pageUrl(`/${i.slug}-video-downloader/`, copy.locale)}
                className="rounded-pill border border-line bg-sand px-3 py-1.5 text-xs font-bold text-ink no-underline hover:border-[#56604F]"

三处要注意的（都是会静默出错的那种）：

facebook-video 这个 slug 我按你现有路径写的吗没有核实 —— 你测试站的 Facebook 页实测是 /facebook-video-downloader/，而 RelatedLinks 里我拼的是 /${slug}-video-downloader/，所以英文页那一条要传 slug: "facebook-video"（已按此写），而 PT/ES 的 Facebook 页如果实际是 /pt/facebook-video-downloader/，那 slug: "facebook" 就会拼成不存在的 /pt/facebook-video-downloader/。你拷完先跑一遍 curl -o /dev/null -w "%{http_code}" 逐条验，或者把 RelatedLinks 改成直接收完整 href。这个组件我刻意不做兜底——上一轮教训就是"链接写了但页面不存在"。
sitemap 要加三条：/instagram-reels-downloader/（可能已有）、/pt/instagram-reels-downloader/、/es/instagram-reels-downloader/。加上上一轮发现的 Twitter 两条缺失，sitemap.ts 这次一并改。
